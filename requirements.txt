# ── Google Sign-In ────────────────────────────

google-auth==2.37.0

# ── Payments ───────────────────────────────────
# razorpay's client imports pkg_resources (part of setuptools) at module load time, using
# the old pre-importlib.metadata API. Newer Python/pip no longer bundle setuptools into a
# fresh venv by default, so without an explicit dependency here the import crashes with
# "ModuleNotFoundError: No module named 'pkg_resources'" the moment anything touches
# payments.py. An UNPINNED setuptools isn't enough either — setuptools' own newer
# releases have been actively stripping pkg_resources out as they deprecate it, so
# pinning to a known version that still bundles it is required, not just "installed".
setuptools==75.6.0
razorpay==1.4.2
