# ── MongoDB ───────────────────────────────────
pymongo==3.11.4
# pymongo needs dnspython at runtime to resolve mongodb+srv:// URIs (MongoDB Atlas's
# standard connection string format) — it's never imported directly by app code, only
# by pymongo internally when it sees the srv:// scheme, so a static import audit of
# gaming_project.main won't catch it missing. Confirmed live on Render: without it,
# every DB-touching request 500s with "The dnspython module must be installed to use
# mongodb+srv:// URIs" — it only worked in local dev because dnspython happened to
# already be installed from something else, undeclared.
dnspython==2.7.0

# ── Auth / crypto ─────────────────────────────
PyJWT==2.10.1
argon2-cffi==25.1.0
pycryptodome==3.21.0
pyotp==2.9.0
qrcode==8.2
# qrcode's default image factory (used by auth_handler.py's TOTP QR generation —
# qr.make_image(...).save(buffer, format="PNG")) is backed by Pillow. Nothing imports
# PIL directly, so this is invisible to any static import scan too, and only fails at
# the moment someone actually triggers QR generation, not at app startup.
Pillow==11.2.1

# ── Google Sign-In ────────────────────────────
# Only google.oauth2.id_token / google.auth.transport.requests are used (verifying an
# already-issued ID token client-side Google Sign-In hands us) — google-auth-oauthlib
# provides the full OAuth2 authorization-code flow (google_auth_oauthlib.flow), which
# this app never uses. Confirmed via grep: zero imports anywhere in the codebase.
google-auth==2.37.0
